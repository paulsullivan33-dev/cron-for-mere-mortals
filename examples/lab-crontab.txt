# cron lab crontab. Every job fires every minute so the lab finishes fast.
# Heartbeat: proof of firing, one line per minute.
* * * * * echo "tick $(date -Is)" >> /home/you/cron-lab/out/heartbeat.log

# What does a cron job's world look like? Dump it.
* * * * * { echo "PWD=$PWD"; echo "SHELL=$SHELL"; echo "PATH=$PATH"; echo "HOME=$HOME"; echo "LOGNAME=$LOGNAME"; /usr/bin/env | sort; } > /home/you/cron-lab/out/env-dump.txt 2>&1

# The percent trap, escaped: the date lands in the file.
* * * * * /bin/date >> /home/you/cron-lab/out/percent-ok.log 2>&1 +\%Y-\%m-\%d
# The percent trap, unescaped: everything from the first % is the command's stdin.
* * * * * /bin/date >> /home/you/cron-lab/out/percent-trap.log 2>&1 +%Y-%m-%d

# A noisy job with no redirect: its output is headed for the mail.
* * * * * echo "noisy job: this output has nowhere to go"

# Overlap: a 75-second job on a 60-second schedule, guarded by flock -n.
* * * * * /usr/bin/flock -n /home/you/cron-lab/out/slow.lock /home/you/cron-lab/slow-job.sh || echo "skipped at $(date -Is) - previous run still going" >> /home/you/cron-lab/out/flock.log

# @reboot: runs when the cron daemon (re)starts, not when a clock ticks.
@reboot echo "reboot job fired at $(date -Is)" >> /home/you/cron-lab/out/reboot.log
